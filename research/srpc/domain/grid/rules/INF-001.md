(function (global) {

const GridRules = {

version: "0.1.0-GRID-RESEARCH",

domain: "grid",

rules: [

  {
    id: "GRID-RULE-001",
    name: "Grid Availability",
    description: "Verify that grid availability remains within the defined operating range.",

    evaluate: function (inputs) {
      const value = inputs.systemState.availability;

      return {
        ruleId: "GRID-RULE-001",
        passed: value >= 0 && value <= 100,
        value: value,
        assessment: value >= 80 ? "STABLE" : "DEGRADED"
      };
    }
  },

  {
    id: "GRID-RULE-002",
    name: "Grid Integrity",
    description: "Verify that system integrity remains within the defined operating range.",

    evaluate: function (inputs) {
      const value = inputs.systemState.integrity;

      return {
        ruleId: "GRID-RULE-002",
        passed: value >= 0 && value <= 100,
        value: value,
        assessment: value >= 80 ? "STABLE" : "DEGRADED"
      };
    }
  },

  {
    id: "GRID-RULE-003",
    name: "Grid Stability",
    description: "Verify that grid stability remains within the defined operating range.",

    evaluate: function (inputs) {
      const value = inputs.systemState.stability;

      return {
        ruleId: "GRID-RULE-003",
        passed: value >= 0 && value <= 100,
        value: value,
        assessment: value >= 80 ? "STABLE" : "DEGRADED"
      };
    }
  },

  {
    id: "GRID-RULE-004",
    name: "Grid Load",
    description: "Assess current grid loading against the defined 0–100 percent range.",

    evaluate: function (inputs) {
      const value = inputs.energy.gridLoadPercent;

      return {
        ruleId: "GRID-RULE-004",
        passed: value >= 0 && value <= 100,
        value: value,
        assessment:
          value >= 90
            ? "HIGH_LOAD"
            : value >= 75
            ? "ELEVATED_LOAD"
            : "NORMAL_LOAD"
      };
    }
  },

  {
    id: "GRID-RULE-005",
    name: "Renewable Energy Contribution",
    description: "Verify that renewable energy contribution remains within the defined range.",

    evaluate: function (inputs) {
      const value = inputs.energy.renewablePercent;

      return {
        ruleId: "GRID-RULE-005",
        passed: value >= 0 && value <= 100,
        value: value,
        assessment:
          value >= 50
            ? "HIGH_RENEWABLE_CONTRIBUTION"
            : "LOWER_RENEWABLE_CONTRIBUTION"
      };
    }
  },

  {
    id: "GRID-RULE-006",
    name: "Carbon Intensity",
    description: "Observe carbon intensity as a grid resilience input without directly controlling the physical grid.",

    evaluate: function (inputs) {
      const value = inputs.energy.carbonIntensity;

      return {
        ruleId: "GRID-RULE-006",
        passed: value >= 0 && value <= 100,
        value: value,
        assessment:
          value <= 40
            ? "LOW"
            : value <= 70
            ? "MODERATE"
            : "HIGH"
      };
    }
  },

  {
    id: "GRID-RULE-007",
    name: "INF-001 Regional Network Outage",
    description:
      "Evaluate regional network connectivity failure and its potential cascading impact across Infrastructure, Data Centre, Cyber, and Financial domains.",

    ruleSource: {
      ruleId: "INF-001",
      domain: "Infrastructure Resilience",
      category: "Network Connectivity Failure",
      status: "Active",
      version: "1.0",
      repository: "Sextant Rule Library"
    },

    riskLevels: {
      GREEN: {
        condition: "Full network connectivity is stable.",
        action: "No contingency actions required."
      },

      YELLOW: {
        condition: "Early signs of network instability.",
        indicators: [
          "Increased latency",
          "Intermittent packet loss",
          "DNS delays"
        ],
        action:
          "Increase monitoring of routing and connectivity layers."
      },

      ORANGE: {
        condition: "Significant regional network degradation.",
        indicators: [
          "Partial regional outages",
          "High latency and packet loss",
          "DNS resolution instability",
          "Service accessibility disruption"
        ],
        cascade: [
          "Network Instability",
          "Service Degradation",
          "Inter-System Communication Failure",
          "Operational Impact"
        ],
        action:
          "Activate redundant routing paths and reroute traffic through stable regions."
      },

      RED: {
        condition: "Critical regional network outage.",
        indicators: [
          "Complete loss of regional connectivity",
          "Severe service unavailability",
          "Routing system collapse",
          "Cross-region communication failure"
        ],
        cascade: [
          "Network Outage",
          "Infrastructure Isolation",
          "Service Disruption",
          "Cross-Domain Systemic Risk"
        ],
        action:
          "Activate emergency rerouting, failover to backup regions, and restore connectivity via alternative backbone providers."
      }
    },

    crossDomainDependencies: [
      "INF",
      "DC",
      "CYB",
      "FIN"
    ],

    primaryIndicators: [
      "Packet loss across regions",
      "DNS resolution failures",
      "Increased latency or timeouts",
      "Inter-data-centre connectivity loss",
      "BGP routing instability",
      "ISP backbone degradation",
      "Service endpoint unavailability"
    ],

    contingencyActions: [
      "Activate redundant network paths immediately",
      "Failover to secondary regions",
      "Monitor routing table stability",
      "Validate DNS integrity",
      "Coordinate with ISP providers",
      "Escalate if outage persists or expands"
    ],

    evaluate: function (inputs) {

      const network =
        inputs.network ||
        inputs.infrastructure ||
        {};

      const connectivity =
        network.connectivityPercent !== undefined
          ? network.connectivityPercent
          : 100;

      const packetLoss =
        network.packetLossPercent !== undefined
          ? network.packetLossPercent
          : 0;

      const dnsFailures =
        network.dnsFailures !== undefined
          ? network.dnsFailures
          : 0;

      const latency =
        network.latencyMs !== undefined
          ? network.latencyMs
          : 0;

      let riskLevel = "GREEN";

      if (
        connectivity < 25 ||
        packetLoss >= 50 ||
        dnsFailures >= 50
      ) {
        riskLevel = "RED";
      }
      else if (
        connectivity < 60 ||
        packetLoss >= 20 ||
        dnsFailures >= 20 ||
        latency >= 500
      ) {
        riskLevel = "ORANGE";
      }
      else if (
        connectivity < 90 ||
        packetLoss > 0 ||
        dnsFailures > 0 ||
        latency >= 150
      ) {
        riskLevel = "YELLOW";
      }

      return {
        ruleId: "INF-001",
        gridRuleId: "GRID-RULE-007",
        passed: riskLevel === "GREEN",
        riskLevel: riskLevel,
        connectivityPercent: connectivity,
        packetLossPercent: packetLoss,
        dnsFailures: dnsFailures,
        latencyMs: latency,
        affectedDomains: [
          "INF",
          "DC",
          "CYB",
          "FIN"
        ],
        cascadePath:
          riskLevel === "RED"
            ? [
                "Network Outage",
                "Infrastructure Isolation",
                "Service Disruption",
                "Cross-Domain Systemic Risk"
              ]
            : riskLevel === "ORANGE"
            ? [
                "Network Instability",
                "Service Degradation",
                "Inter-System Communication Failure",
                "Operational Impact"
              ]
            : [],
        recommendedContingencyActions:
          riskLevel === "GREEN"
            ? []
            : riskLevel === "YELLOW"
            ? [
                "Increase monitoring of routing and connectivity layers."
              ]
            : riskLevel === "ORANGE"
            ? [
                "Activate redundant routing paths.",
                "Reroute traffic through stable regions."
              ]
            : [
                "Activate emergency rerouting.",
                "Failover to backup regions.",
                "Restore connectivity via alternative backbone providers."
              ],
        humanAuthorizationRequired: true,
        physicalExecution: false,
        autonomousActuation: false
      };
    }
  }

],

getRules: function () {
  return this.rules;
},

getRuleCount: function () {
  return this.rules.length;
}

};

global.SextantGridRules = GridRules;

})(window);
